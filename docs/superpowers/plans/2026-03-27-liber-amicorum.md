- [ ] **Step 4: Write the extraction script**

Create `scripts/extract_data.py`. This script:
1. Loads `data/dprr.ttl` into an ephemeral pyoxigraph Store
2. Loads the prefix map from `dprr_mcp/context/prefixes.yaml`
3. Runs SPARQL queries to extract persons (with all nested assertions) and offices (with holders)
4. Writes `data/extracted/persons.json`, `data/extracted/offices.json`, `data/extracted/search-index.json`

Key implementation notes:
- Use `pyoxigraph.Store()` (in-memory, no path) for ephemeral use
- Use `store.bulk_load()` to load the TTL file
- Use `store.query()` for SPARQL SELECT queries
- The prefix map from `dprr_mcp/context/prefixes.yaml` provides the namespace URIs
- Build prefix headers from the map for SPARQL queries
- Office IDs: `slugify_office(name)` lowercases and replaces spaces with hyphens
- Person IDs: extract `vocab:hasDprrID` value directly
- For each person, run sub-queries to collect post assertions, relationships, status assertions, date information, and tribe assertions
- For each office, run a query to collect all holders via PostAssertion joins
- Boolean fields (patrician, nobilis, novus, uncertainty flags): default to `False` if the triple doesn't exist in the store

The script should be runnable as `uv run python scripts/extract_data.py` and also importable for testing.

```python
"""Extract DPRR RDF data into JSON for the liber-amicorum static site."""

from __future__ import annotations

import json
import sys
from pathlib import Path

import yaml
from pyoxigraph import Store


ROOT = Path(__file__).resolve().parent.parent
DATA_DIR = ROOT / "data"
TTL_PATH = DATA_DIR / "dprr.ttl"
OUTPUT_DIR = DATA_DIR / "extracted"
PREFIXES_PATH = ROOT / "dprr_mcp" / "context" / "prefixes.yaml"


def load_prefix_map() -> dict[str, str]:
    """Load namespace prefix map from the shared YAML config."""
    with open(PREFIXES_PATH) as f:
        return yaml.safe_load(f)["prefixes"]


def sparql_prefixes(prefix_map: dict[str, str]) -> str:
    """Build SPARQL PREFIX header from prefix map."""
    return "\n".join(f"PREFIX {k}: <{v}>" for k, v in prefix_map.items())


def query_to_dicts(store: Store, sparql: str) -> list[dict]:
    """Execute a SPARQL SELECT and return results as list of dicts."""
    results = store.query(sparql)
    rows = []
    for solution in results:
        row = {}
        for var in solution:
            val = solution[var]
            if val is None:
                row[var] = None
            else:
                row[var] = val.value
        rows.append(row)
    return rows


def slugify_office(name: str) -> str:
    """Convert office name to URL-safe slug."""
    return name.lower().replace(" ", "-")


def to_bool(val: str | None) -> bool:
    """Convert RDF boolean string to Python bool."""
    return val is not None and val.lower() == "true"


def to_int(val: str | None) -> int | None:
    """Convert RDF integer string to Python int."""
    if val is None:
        return None
    try:
        return int(val)
    except (ValueError, TypeError):
        return None


def extract_persons(store: Store, prefix_map: dict[str, str]) -> list[dict]:
    """Extract all persons with nested assertions."""
    pfx = sparql_prefixes(prefix_map)

    # 1. Get all persons with basic properties
    person_query = pfx + """
    SELECT ?person ?dprrId ?name ?praenomen ?praenomenName ?nomen ?cognomen
           ?filiation ?otherNames ?eraFrom ?eraTo ?sex ?sexLabel
           ?patrician ?nobilis ?novus ?highestOffice ?reNumber ?origin
           ?praenomenUncertain ?nomenUncertain ?cognomenUncertain
           ?filiationUncertain ?otherNamesUncertain ?patricianUncertain
           ?nobilisNotes ?novusNotes
    WHERE {
        ?person a vocab:Person .
        OPTIONAL { ?person vocab:hasDprrID ?dprrId }
        OPTIONAL { ?person vocab:hasPersonName ?name }
        OPTIONAL {
            ?person vocab:hasPraenomen ?praenomen .
            ?praenomen vocab:hasName ?praenomenName .
        }
        OPTIONAL { ?person vocab:hasNomen ?nomen }
        OPTIONAL { ?person vocab:hasCognomen ?cognomen }
        OPTIONAL { ?person vocab:hasFiliation ?filiation }
        OPTIONAL { ?person vocab:hasOtherNames ?otherNames }
        OPTIONAL { ?person vocab:hasEraFrom ?eraFrom }
        OPTIONAL { ?person vocab:hasEraTo ?eraTo }
        OPTIONAL {
            ?person vocab:isSex ?sex .
            ?sex rdfs:label ?sexLabel .
        }
        OPTIONAL { ?person vocab:isPatrician ?patrician }
        OPTIONAL { ?person vocab:isNobilis ?nobilis }
        OPTIONAL { ?person vocab:isNovus ?novus }
        OPTIONAL { ?person vocab:hasHighestOffice ?highestOffice }
        OPTIONAL { ?person vocab:hasReNumber ?reNumber }
        OPTIONAL { ?person vocab:hasOrigin ?origin }
        OPTIONAL { ?person vocab:isPraenomenUncertain ?praenomenUncertain }
        OPTIONAL { ?person vocab:isNomenUncertain ?nomenUncertain }
        OPTIONAL { ?person vocab:isCognomenUncertain ?cognomenUncertain }
        OPTIONAL { ?person vocab:isFiliationUncertain ?filiationUncertain }
        OPTIONAL { ?person vocab:isOtherNamesUncertain ?otherNamesUncertain }
        OPTIONAL { ?person vocab:isPatricianUncertain ?patricianUncertain }
        OPTIONAL { ?person vocab:hasNobilisNotes ?nobilisNotes }
        OPTIONAL { ?person vocab:hasNovusNotes ?novusNotes }
    }
    """
    raw_persons = query_to_dicts(store, person_query)

    # Deduplicate by person URI (multiple rows from OPTIONAL joins)
    person_map: dict[str, dict] = {}
    for row in raw_persons:
        uri = row["person"]
        if uri not in person_map:
            sex_label = row.get("sexLabel", "")
            sex_clean = sex_label.replace("Sex: ", "") if sex_label else None
            person_map[uri] = {
                "id": row.get("dprrId"),
                "name": row.get("name"),
                "praenomen": row.get("praenomenName"),
                "nomen": row.get("nomen"),
                "cognomen": row.get("cognomen"),
                "filiation": row.get("filiation"),
                "eraFrom": to_int(row.get("eraFrom")),
                "eraTo": to_int(row.get("eraTo")),
                "sex": sex_clean,
                "patrician": to_bool(row.get("patrician")),
                "nobilis": to_bool(row.get("nobilis")),
                "novus": to_bool(row.get("novus")),
                "highestOffice": row.get("highestOffice"),
                "reNumber": row.get("reNumber"),
                "origin": row.get("origin"),
                "praenomenUncertain": to_bool(row.get("praenomenUncertain")),
                "nomenUncertain": to_bool(row.get("nomenUncertain")),
                "cognomenUncertain": to_bool(row.get("cognomenUncertain")),
                "filiationUncertain": to_bool(row.get("filiationUncertain")),
                "otherNamesUncertain": to_bool(row.get("otherNamesUncertain")),
                "patricianUncertain": to_bool(row.get("patricianUncertain")),
                "notes": [],
                "postAssertions": [],
                "relationships": [],
                "statusAssertions": [],
                "dateInformation": [],
                "tribeAssertions": [],
                "_uri": uri,
            }

    # 2. Person notes
    notes_query = pfx + """
    SELECT ?person ?noteText
    WHERE {
        ?person a vocab:Person ;
                vocab:hasPersonNote ?note .
        ?note vocab:hasNoteText ?noteText .
    }
    """
    for row in query_to_dicts(store, notes_query):
        uri = row["person"]
        if uri in person_map and row.get("noteText"):
            person_map[uri]["notes"].append(row["noteText"])

    # 3. Post assertions
    pa_query = pfx + """
    SELECT ?person ?pa ?officeName ?officeUri ?dateStart ?dateEnd
           ?uncertain ?dateStartUncertain ?dateEndUncertain
           ?sourceName ?sourceUri ?dateSourceName ?dateSourceUri
           ?noteText ?originalText ?provinceName ?provinceUncertain
    WHERE {
        ?pa a vocab:PostAssertion ;
            vocab:isAboutPerson ?person ;
            vocab:hasOffice ?officeUri .
        ?officeUri vocab:hasName ?officeName .
        OPTIONAL { ?pa vocab:hasDateStart ?dateStart }
        OPTIONAL { ?pa vocab:hasDateEnd ?dateEnd }
        OPTIONAL { ?pa vocab:isUncertain ?uncertain }
        OPTIONAL { ?pa vocab:isDateStartUncertain ?dateStartUncertain }
        OPTIONAL { ?pa vocab:isDateEndUncertain ?dateEndUncertain }
        OPTIONAL {
            ?pa vocab:hasSecondarySource ?sourceUri .
            ?sourceUri vocab:hasAbbreviation ?sourceName .
        }
        OPTIONAL {
            ?pa vocab:hasDateSecondarySource ?dateSourceUri .
            ?dateSourceUri vocab:hasAbbreviation ?dateSourceName .
        }
        OPTIONAL {
            ?pa vocab:hasPostAssertionNote ?paNote .
            ?paNote vocab:hasNoteText ?noteText .
        }
        OPTIONAL { ?pa vocab:hasOriginalText ?originalText }
        OPTIONAL {
            ?pap a vocab:PostAssertionProvince ;
                 vocab:hasPostAssertion ?pa ;
                 vocab:hasProvince ?prov .
            ?prov vocab:hasName ?provinceName .
            OPTIONAL { ?pap vocab:isProvinceUncertain ?provinceUncertain }
        }
    }
    """
    # Group post assertion rows by PA URI (multiple sources/notes per assertion)
    pa_groups: dict[str, dict] = {}
    for row in query_to_dicts(store, pa_query):
        pa_uri = row["pa"]
        person_uri = row["person"]
        if pa_uri not in pa_groups:
            pa_groups[pa_uri] = {
                "personUri": person_uri,
                "office": row.get("officeName"),
                "officeId": slugify_office(row["officeName"]) if row.get("officeName") else None,
                "dateStart": to_int(row.get("dateStart")),
                "dateEnd": to_int(row.get("dateEnd")),
                "province": row.get("provinceName"),
                "provinceUncertain": to_bool(row.get("provinceUncertain")),
                "uncertain": to_bool(row.get("uncertain")),
                "dateStartUncertain": to_bool(row.get("dateStartUncertain")),
                "dateEndUncertain": to_bool(row.get("dateEndUncertain")),
                "sources": {},
                "dateSource": None,
                "notes": set(),
                "originalText": row.get("originalText"),
            }
        group = pa_groups[pa_uri]
        if row.get("sourceName") and row.get("sourceUri"):
            group["sources"][row["sourceUri"]] = {
                "name": row["sourceName"],
                "id": slugify_office(row["sourceName"]),
            }
        if row.get("dateSourceName"):
            group["dateSource"] = {
                "name": row["dateSourceName"],
                "id": slugify_office(row["dateSourceName"]),
            }
        if row.get("noteText"):
            group["notes"].add(row["noteText"])

    for pa_uri, group in pa_groups.items():
        person_uri = group.pop("personUri")
        group["sources"] = list(group["sources"].values())
        group["notes"] = list(group["notes"])
        if person_uri in person_map:
            person_map[person_uri]["postAssertions"].append(group)

    # 4. Relationships
    rel_query = pfx + """
    SELECT ?person ?relatedPerson ?relatedDprrId ?relatedName ?relType ?relLabel
    WHERE {
        ?ra a vocab:RelationshipAssertion ;
            vocab:isAboutPerson ?person ;
            vocab:hasRelatedPerson ?relatedPerson ;
            vocab:hasRelationship ?relType .
        ?relType rdfs:label ?relLabel .
        OPTIONAL { ?relatedPerson vocab:hasDprrID ?relatedDprrId }
        OPTIONAL { ?relatedPerson vocab:hasPersonName ?relatedName }
    }
    """
    for row in query_to_dicts(store, rel_query):
        uri = row["person"]
        if uri in person_map:
            rel_label = row.get("relLabel", "")
            rel_type = rel_label.replace("Relationship: ", "") if rel_label else rel_label
            person_map[uri]["relationships"].append({
                "type": rel_type,
                "relatedPersonId": row.get("relatedDprrId"),
                "relatedPersonName": row.get("relatedName"),
            })

    # 5. Status assertions
    sa_query = pfx + """
    SELECT ?person ?statusLabel ?dateStart ?dateEnd
           ?uncertain ?dateStartUncertain ?dateEndUncertain
           ?sourceName ?sourceUri ?noteText ?originalText
    WHERE {
        ?sa a vocab:StatusAssertion ;
            vocab:isAboutPerson ?person ;
            vocab:hasStatus ?status .
        ?status rdfs:label ?statusLabel .
        OPTIONAL { ?sa vocab:hasDateStart ?dateStart }
        OPTIONAL { ?sa vocab:hasDateEnd ?dateEnd }
        OPTIONAL { ?sa vocab:isUncertain ?uncertain }
        OPTIONAL { ?sa vocab:isDateStartUncertain ?dateStartUncertain }
        OPTIONAL { ?sa vocab:isDateEndUncertain ?dateEndUncertain }
        OPTIONAL {
            ?sa vocab:hasSecondarySource ?sourceUri .
            ?sourceUri vocab:hasAbbreviation ?sourceName .
        }
        OPTIONAL {
            ?sa vocab:hasStatusAssertionNote ?saNote .
            ?saNote vocab:hasNoteText ?noteText .
        }
        OPTIONAL { ?sa vocab:hasOriginalText ?originalText }
    }
    """
    for row in query_to_dicts(store, sa_query):
        uri = row["person"]
        if uri in person_map:
            status_label = row.get("statusLabel", "")
            status_clean = status_label.replace("Status: ", "") if status_label else status_label
            sources = []
            if row.get("sourceName"):
                sources.append({
                    "name": row["sourceName"],
                    "id": slugify_office(row["sourceName"]),
                })
            notes = []
            if row.get("noteText"):
                notes.append(row["noteText"])
            person_map[uri]["statusAssertions"].append({
                "status": status_clean,
                "dateStart": to_int(row.get("dateStart")),
                "dateEnd": to_int(row.get("dateEnd")),
                "uncertain": to_bool(row.get("uncertain")),
                "dateStartUncertain": to_bool(row.get("dateStartUncertain")),
                "dateEndUncertain": to_bool(row.get("dateEndUncertain")),
                "sources": sources,
                "notes": notes,
                "originalText": row.get("originalText"),
            })

    # 6. Date information
    di_query = pfx + """
    SELECT ?person ?dateTypeLabel ?value ?interval ?uncertain
           ?sourceName ?sourceUri ?notes
    WHERE {
        ?di a vocab:DateInformation ;
            vocab:isAboutPerson ?person ;
            vocab:hasDateType ?dateType .
        ?dateType rdfs:label ?dateTypeLabel .
        OPTIONAL { ?di vocab:hasValue ?value }
        OPTIONAL { ?di vocab:hasDateInterval ?interval }
        OPTIONAL { ?di vocab:isUncertain ?uncertain }
        OPTIONAL {
            ?di vocab:hasSecondarySource ?sourceUri .
            ?sourceUri vocab:hasAbbreviation ?sourceName .
        }
        OPTIONAL { ?di vocab:hasNotes ?notes }
    }
    """
    for row in query_to_dicts(store, di_query):
        uri = row["person"]
        if uri in person_map:
            type_label = row.get("dateTypeLabel", "")
            type_clean = type_label.replace("Date Type: ", "") if type_label else type_label
            source = None
            if row.get("sourceName"):
                source = {
                    "name": row["sourceName"],
                    "id": slugify_office(row["sourceName"]),
                }
            person_map[uri]["dateInformation"].append({
                "type": type_clean,
                "value": to_int(row.get("value")),
                "interval": row.get("interval"),
                "uncertain": to_bool(row.get("uncertain")),
                "source": source,
            })

    # 7. Tribe assertions
    ta_query = pfx + """
    SELECT ?person ?tribeLabel ?uncertain ?sourceName ?sourceUri
    WHERE {
        ?ta a vocab:TribeAssertion ;
            vocab:isAboutPerson ?person ;
            vocab:hasTribe ?tribe .
        ?tribe rdfs:label ?tribeLabel .
        OPTIONAL { ?ta vocab:isUncertain ?uncertain }
        OPTIONAL {
            ?ta vocab:hasSecondarySource ?sourceUri .
            ?sourceUri vocab:hasAbbreviation ?sourceName .
        }
    }
    """
    for row in query_to_dicts(store, ta_query):
        uri = row["person"]
        if uri in person_map:
            tribe_label = row.get("tribeLabel", "")
            tribe_clean = tribe_label.replace("Tribe: ", "") if tribe_label else tribe_label
            source = None
            if row.get("sourceName"):
                source = {
                    "name": row["sourceName"],
                    "id": slugify_office(row["sourceName"]),
                }
            person_map[uri]["tribeAssertions"].append({
                "tribe": tribe_clean,
                "uncertain": to_bool(row.get("uncertain")),
                "source": source,
            })

    # Clean up internal fields and return
    persons = []
    for p in person_map.values():
        del p["_uri"]
        persons.append(p)

    # Sort by nomen then cognomen for stable output
    persons.sort(key=lambda p: (p.get("nomen") or "", p.get("cognomen") or "", p.get("id") or ""))
    return persons


def extract_offices(store: Store, prefix_map: dict[str, str]) -> list[dict]:
    """Extract all offices with holders."""
    pfx = sparql_prefixes(prefix_map)

    # Get all offices
    office_query = pfx + """
    SELECT ?office ?name ?abbreviation ?parentUri ?parentName
    WHERE {
        ?office a vocab:Office .
        OPTIONAL { ?office vocab:hasName ?name }
        OPTIONAL { ?office vocab:hasAbbreviation ?abbreviation }
        OPTIONAL {
            ?office vocab:hasParent ?parentUri .
            ?parentUri vocab:hasName ?parentName .
        }
    }
    """
    raw_offices = query_to_dicts(store, office_query)

    office_map: dict[str, dict] = {}
    for row in raw_offices:
        uri = row["office"]
        if uri not in office_map:
            office_map[uri] = {
                "id": slugify_office(row["name"]) if row.get("name") else uri,
                "name": row.get("name"),
                "abbreviation": row.get("abbreviation"),
                "parentId": slugify_office(row["parentName"]) if row.get("parentName") else None,
                "parentName": row.get("parentName"),
                "children": [],
                "holders": [],
                "_uri": uri,
            }

    # Build children from parent relationships
    for uri, office in office_map.items():
        if office["parentId"]:
            parent = next(
                (o for o in office_map.values() if o["id"] == office["parentId"]),
                None,
            )
            if parent:
                parent["children"].append({"id": office["id"], "name": office["name"]})

    # Get holders via PostAssertion
    holders_query = pfx + """
    SELECT ?officeUri ?personDprrId ?personName ?dateStart ?dateEnd
           ?uncertain ?dateStartUncertain ?dateEndUncertain
           ?sourceName ?sourceUri
    WHERE {
        ?pa a vocab:PostAssertion ;
            vocab:isAboutPerson ?person ;
            vocab:hasOffice ?officeUri .
        ?person vocab:hasDprrID ?personDprrId .
        OPTIONAL { ?person vocab:hasPersonName ?personName }
        OPTIONAL { ?pa vocab:hasDateStart ?dateStart }
        OPTIONAL { ?pa vocab:hasDateEnd ?dateEnd }
        OPTIONAL { ?pa vocab:isUncertain ?uncertain }
        OPTIONAL { ?pa vocab:isDateStartUncertain ?dateStartUncertain }
        OPTIONAL { ?pa vocab:isDateEndUncertain ?dateEndUncertain }
        OPTIONAL {
            ?pa vocab:hasSecondarySource ?sourceUri .
            ?sourceUri vocab:hasAbbreviation ?sourceName .
        }
    }
    """
    # Group by (office, person, dateStart) to deduplicate across multiple sources
    holder_groups: dict[tuple, dict] = {}
    for row in query_to_dicts(store, holders_query):
        office_uri = row["officeUri"]
        key = (office_uri, row.get("personDprrId"), row.get("dateStart"))
        if key not in holder_groups:
            holder_groups[key] = {
                "officeUri": office_uri,
                "personId": row.get("personDprrId"),
                "personName": row.get("personName"),
                "dateStart": to_int(row.get("dateStart")),
                "dateEnd": to_int(row.get("dateEnd")),
                "uncertain": to_bool(row.get("uncertain")),
                "dateStartUncertain": to_bool(row.get("dateStartUncertain")),
                "dateEndUncertain": to_bool(row.get("dateEndUncertain")),
                "sources": {},
            }
        if row.get("sourceName") and row.get("sourceUri"):
            holder_groups[key]["sources"][row["sourceUri"]] = {
                "name": row["sourceName"],
                "id": slugify_office(row["sourceName"]),
            }

    for group in holder_groups.values():
        office_uri = group.pop("officeUri")
        group["sources"] = list(group["sources"].values())
        if office_uri in office_map:
            office_map[office_uri]["holders"].append(group)

    # Sort holders by date
    for office in office_map.values():
        office["holders"].sort(key=lambda h: h.get("dateStart") or 0)

    # Clean up and return
    offices = []
    for o in office_map.values():
        del o["_uri"]
        offices.append(o)
    offices.sort(key=lambda o: o.get("name") or "")
    return offices


def build_search_index(persons: list[dict], offices: list[dict]) -> list[dict]:
    """Build a flat search index for MiniSearch."""
    index = []
    for p in persons:
        era_parts = []
        if p.get("eraFrom") is not None:
            era_parts.append(str(p["eraFrom"]))
        if p.get("eraTo") is not None:
            era_parts.append(str(p["eraTo"]))
        index.append({
            "id": p["id"],
            "type": "person",
            "name": p.get("name"),
            "nomen": p.get("nomen"),
            "cognomen": p.get("cognomen"),
            "praenomen": p.get("praenomen"),
            "highestOffice": p.get("highestOffice"),
            "era": " to ".join(era_parts) if era_parts else None,
        })
    for o in offices:
        index.append({
            "id": o["id"],
            "type": "office",
            "name": o.get("name"),
            "abbreviation": o.get("abbreviation"),
        })
    return index
```

```ts
/**
 * Format a year value for display. Negative = BC, positive = AD.
 * Returns null if value is null/undefined.
 */
export function formatYear(value: number | null | undefined): string | null {
  if (value == null) return null
  if (value < 0) return `${Math.abs(value)} BC`
  return `${value} AD`
}

/**
 * Format a date range (eraFrom - eraTo).
 */
export function formatEra(from: number | null, to: number | null): string {
  const f = formatYear(from)
  const t = formatYear(to)
  if (f && t && f !== t) return `${f} – ${t}`
  if (f) return f
  if (t) return t
  return "unknown"
}

/**
 * Format a year with uncertainty marker.
 */
export function formatYearUncertain(
  value: number | null,
  uncertain: boolean,
): string | null {
  const formatted = formatYear(value)
  if (!formatted) return null
  return uncertain ? `${formatted}?` : formatted
}
```

```tsx
  children: React.ReactNode
  source?: string
}

export function UncertaintyMarker({
  uncertain,
  children,
  source,
}: UncertaintyMarkerProps) {
  if (!uncertain) return <>{children}</>

  const content = (
    <span className="border-b border-dotted border-uncertain italic">
      {children}
      <span className="text-uncertain">?</span>
    </span>
  )

  if (!source) return content

  return (
    <TooltipProvider>
      <Tooltip>
        <TooltipTrigger asChild>{content}</TooltipTrigger>
        <TooltipContent>
          <p className="text-xs">Uncertain — {source}</p>
        </TooltipContent>
      </Tooltip>
    </TooltipProvider>
  )
}
```

```tsx
import { createFileRoute, Link } from "@tanstack/react-router"
import { getAllOffices } from "#/lib/data"

export const Route = createFileRoute("/offices/")({
  loader: () => getAllOffices(),
  component: OfficesIndex,
})

function OfficesIndex() {
  const offices = Route.useLoaderData()

  // Group offices by parent (top-level offices have parentId === null)
  const topLevel = offices.filter((o) => !o.parentId)
  const byParent = new Map<string, typeof offices>()
  for (const o of offices) {
    if (o.parentId) {
      const group = byParent.get(o.parentId) ?? []
      group.push(o)
      byParent.set(o.parentId, group)
    }
  }

  return (
```

```ts
import MiniSearch from "minisearch"
import type { SearchDoc } from "./types"
import { getSearchIndex } from "./data"

let searchInstance: MiniSearch<SearchDoc> | null = null

export function getSearch(): MiniSearch<SearchDoc> {
  if (searchInstance) return searchInstance

  searchInstance = new MiniSearch<SearchDoc>({
    fields: ["name", "nomen", "cognomen", "praenomen", "abbreviation"],
    storeFields: ["id", "type", "name", "nomen", "highestOffice", "era", "abbreviation"],
    searchOptions: {
      boost: { nomen: 2, cognomen: 2 },
      fuzzy: 0.2,
      prefix: true,
    },
  })

  searchInstance.addAll(getSearchIndex())
  return searchInstance
}

export interface SearchResult {
  id: string
  type: "person" | "office"
  name: string
  nomen?: string
  highestOffice?: string
  era?: string
  abbreviation?: string
}

export function search(query: string): SearchResult[] {
  if (!query.trim()) return []
  return getSearch().search(query) as unknown as SearchResult[]
}
```

A command palette-style search in the header using shadcn/ui Popover + Command:

```tsx
import { useState, useCallback, useRef, useEffect } from "react"
import { useNavigate } from "@tanstack/react-router"
import { search, type SearchResult } from "#/lib/search"
import { Input } from "#/components/ui/input"
import {
  Popover,
  PopoverContent,
  PopoverTrigger,
} from "#/components/ui/popover"

export function SearchCommand() {
  const [query, setQuery] = useState("")
  const [results, setResults] = useState<SearchResult[]>([])
  const [open, setOpen] = useState(false)
  const [selectedIndex, setSelectedIndex] = useState(0)
  const navigate = useNavigate()
  const debounceRef = useRef<ReturnType<typeof setTimeout>>()

  const handleSearch = useCallback((value: string) => {
    setQuery(value)
    if (debounceRef.current) clearTimeout(debounceRef.current)
    debounceRef.current = setTimeout(() => {
      const res = search(value)
      setResults(res.slice(0, 10))
      setSelectedIndex(0)
      setOpen(res.length > 0 && value.length > 0)
    }, 150)
  }, [])

  const navigateToResult = useCallback(
    (result: SearchResult) => {
      setOpen(false)
      setQuery("")
      if (result.type === "person") {
        navigate({ to: "/persons/$personId", params: { personId: result.id } })
      } else {
        navigate({ to: "/offices/$officeId", params: { officeId: result.id } })
      }
    },
    [navigate],
  )

  const handleKeyDown = useCallback(
    (e: React.KeyboardEvent) => {
      if (e.key === "ArrowDown") {
        e.preventDefault()
        setSelectedIndex((i) => Math.min(i + 1, results.length - 1))
      } else if (e.key === "ArrowUp") {
        e.preventDefault()
        setSelectedIndex((i) => Math.max(i - 1, 0))
      } else if (e.key === "Enter" && results[selectedIndex]) {
        e.preventDefault()
        navigateToResult(results[selectedIndex])
      } else if (e.key === "Escape") {
        setOpen(false)
      }
    },
    [results, selectedIndex, navigateToResult],
  )

  // Close on outside click
  useEffect(() => {
    if (!open) return
    const handler = () => setOpen(false)
    document.addEventListener("click", handler)
    return () => document.removeEventListener("click", handler)
  }, [open])

  const persons = results.filter((r) => r.type === "person")
  const offices = results.filter((r) => r.type === "office")
  let flatIndex = 0

  return (
    <div className="relative" onClick={(e) => e.stopPropagation()}>
      <Popover open={open} onOpenChange={setOpen}>
        <PopoverTrigger asChild>
          <Input
            placeholder="Search persons, offices..."
            value={query}
            onChange={(e) => handleSearch(e.target.value)}
            onKeyDown={handleKeyDown}
            onFocus={() => {
              if (results.length > 0 && query.length > 0) setOpen(true)
            }}
            className="w-64"
          />
        </PopoverTrigger>
        <PopoverContent
          className="w-80 p-0"
          align="start"
          onOpenAutoFocus={(e) => e.preventDefault()}
        >
          <div className="max-h-80 overflow-y-auto">
            {persons.length > 0 && (
              <div>
                <p className="px-3 py-1.5 text-xs font-medium text-ink-muted">
                  Persons
                </p>
                {persons.map((r) => {
                  const idx = flatIndex++
                  return (
                    <button
                      key={r.id}
                      className={`w-full px-3 py-2 text-left text-sm hover:bg-parchment-dark ${
                        idx === selectedIndex ? "bg-parchment-dark" : ""
                      }`}
                      onClick={() => navigateToResult(r)}
                    >
                      <span className="font-medium">{r.name}</span>
                      {r.era && (
                        <span className="ml-2 text-xs text-ink-muted">
                          {r.era}
                        </span>
                      )}
                      {r.highestOffice && (
                        <span className="ml-2 text-xs text-ink-muted">
                          — {r.highestOffice}
                        </span>
                      )}
                    </button>
                  )
                })}
              </div>
            )}
            {offices.length > 0 && (
              <div>
                <p className="px-3 py-1.5 text-xs font-medium text-ink-muted">
                  Offices
                </p>
                {offices.map((r) => {
                  const idx = flatIndex++
                  return (
                    <button
                      key={r.id}
                      className={`w-full px-3 py-2 text-left text-sm hover:bg-parchment-dark ${
                        idx === selectedIndex ? "bg-parchment-dark" : ""
                      }`}
                      onClick={() => navigateToResult(r)}
                    >
                      <span className="font-medium">{r.name}</span>
                      {r.abbreviation && (
                        <span className="ml-2 text-xs text-ink-muted">
                          ({r.abbreviation})
                        </span>
                      )}
                    </button>
                  )
                })}
              </div>
            )}
          </div>
        </PopoverContent>
      </Popover>
    </div>
  )
}
```
